use actix_web::{get, post, web, App, HttpResponse, HttpServer, Responder};
use serde::{Deserialize, Serialize};

// Define the neural node structure
#[derive(Serialize, Deserialize)]
struct NeuralNode {
    id: String,
    activation_level: f32,
    status: String,
}

#[get("/api/v1/status")]
async fn get_status() -> impl Responder {

    let node = NeuralNode {
        id: "core-alpha-01".to_string(),
        activation_level: 0.94,
        status: "optimal".to_string(),
    };

    HttpResponse::Ok().json(node)
}

#[actix_web::main]
async fn main() -> std::io::Result<()> {
    println!("Initializing Neurodeck Workspace on port 8080...");
    
    HttpServer::new(|| {
        App::new()
            .service(get_status)
    })
    .bind(("127.0.0.1", 8080))?
    .run()
    .await
}
